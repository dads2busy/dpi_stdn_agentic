\section{STDN-GEN Architecture}

\begin{figure}[t]
\centering
\begin{mermaid}
graph TB
    Start([CLI Entry]) --> Config[Load Config<br/>& Tech List]
    
    Config --> S1[Stage 1:<br/>Component Extraction]
    
    S1 --> S1Check{Conf > 0.85<br/>& no debate?}
    S1Check -->|No| S1Debate[Multi-Agent Debate<br/>3 agents + Judge]
    S1Check -->|Yes| S2
    S1Debate --> S2
    
    S2[Stage 2:<br/>Materials Mapping] --> S2Filter[Ontology Filter]
    
    S2Filter --> S2Check{>30% filtered<br/>or debate?}
    S2Check -->|Yes| S2Debate[Multi-Agent Debate<br/>3 agents + Judge]
    S2Check -->|No| S3
    S2Debate --> S3
    
    S3[Stage 3:<br/>Country Data] --> S3Query{Material in<br/>USGS DB?}
    
    S3Query -->|Yes| S3USGS[USGS Data<br/>conf=1.0]
    S3Query -->|No| S3LLM[LLM Fallback<br/>conf=0.5-0.8]
    
    S3LLM --> S3Debate{Debate<br/>enabled?}
    S3Debate -->|Yes| S3Val[MCP Tool<br/>Validation]
    S3Debate -->|No| Output
    S3Val --> Output
    S3USGS --> Output
    
    Output[Generate Outputs:<br/>JSON + CSV + Report] --> More{More<br/>techs?}
    
    More -->|Yes| S1
    More -->|No| End([Complete])

    style S2 fill:#fff4e1
    style S3 fill:#e8f5e9
    style S1Debate fill:#ffe1e1
    style S2Debate fill:#ffe1e1
    style S3Val fill:#ffe1e1
\end{mermaid}
